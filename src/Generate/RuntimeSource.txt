#region Services (implement these to run the code)

/// <summary>Runs queries. A text parameter (AddParameter) replaces its [Param.N] in the text by the value as text, so
/// values that go into quotes are escaped by the callers (stringreplace "'" -> "''"). A typed parameter
/// (AddTypedParameter, MII's Use Typed Parameters) is checked against its type and bound as a statement parameter.</summary>
public interface IQueryService
{
    /// <summary>Runs a statement that returns rows and maps the columns to the properties of T by name.</summary>
    List<T> Execute<T>(Query query) where T : MiiRow, new();

    /// <summary>Runs a statement without result rows (query mode Command).</summary>
    void ExecuteCommand(Query query);
}

/// <summary>A query template: the statement text and the parameter values by name (the comment at the call maps
/// [Param.N] to the names).</summary>
public sealed class Query(string text)
{
    public string Text { get; } = text;

    public List<KeyValuePair<string, object>> Parameters { get; } = [];

    /// <summary>MII's RowCount: the query returns at most this many rows and drops the others without a message
    /// (0: not set).</summary>
    public int RowCount { get; set; }

    /// <summary>The MII type (String, Integer, Timestamp, ...) of each typed parameter by name.</summary>
    public Dictionary<string, string> ParameterTypes { get; } = [];

    /// <summary>A text parameter: its [Param.N] is replaced by the value as text.</summary>
    public void AddParameter(string name, object value) => Parameters.Add(new(name, value));

    /// <summary>A typed parameter: checked against the type and bound as a statement parameter.</summary>
    public void AddTypedParameter(string name, string type, object value)
    {
        ParameterTypes[name] = type;
        Parameters.Add(new(name, value));
    }
}

/// <summary>Everything MII does that has no plain C# equivalent: SAP ME PAPI calls, web services, HTTP, mail, files,
/// sort/filter actions, logging, shared properties.</summary>
public interface IPluginService
{
    /// <summary>Runs an MII action: actionType is the MII action class (e.g. SAPMEPAPIInterface), actionName the action
    /// in the transaction, settings its configuration and inputs by MII property name.</summary>
    PluginResult Invoke(string actionType, string actionName, Dictionary<string, object> settings);

    /// <summary>Tracer and Event Logger.</summary>
    void Log(string level, string message);

    /// <summary>Loads a web resource of the MII server (e.g. an XSLT stylesheet) as text.</summary>
    string LoadResource(string path);

    /// <summary>Reads a shared or global property of the MII server (Inline.X, Persistent.X, Global.X).</summary>
    object GetProperty(string name);

    /// <summary>Writes a shared property of the MII server.</summary>
    void SetProperty(string name, object value);
}

/// <summary>The outputs of a plugin call by MII property name ("" for an output that is not set).</summary>
public sealed class PluginResult
{
    public Dictionary<string, object> Values { get; } = [];

    public object Get(string name) => Values.TryGetValue(name, out var value) ? value : string.Empty;
}

/// <summary>An exception raised by a Throw action (or by an action that failed).</summary>
public class MiiException(string message, string actionName, string actionType = "Throw") : Exception(message)
{
    /// <summary>The MII action that raised the exception (Catch output Source).</summary>
    public string ActionName { get; } = actionName;

    /// <summary>The MII action class (Catch output ActionType).</summary>
    public string ActionType { get; } = actionType;

    /// <summary>Catch output ExceptionType: USER_EXCEPTION for a Throw, ACTION_FAILURE for an action that failed
    /// (also when an incoming link failed), OUTPUT_LINK for an outgoing link.</summary>
    public virtual string ExceptionType => ActionType == "Throw" ? "USER_EXCEPTION" : "ACTION_FAILURE";
}

/// <summary>Terminate Transaction with error (or a Throw with TerminateException): ends the transaction with status
/// Failed; a Catch of the same transaction does not stop it, the calling transaction sees it as a failed call. An
/// exception that no Catch takes is different: that transaction ends with status Terminated and its caller goes on.</summary>
public sealed class TransactionTerminatedException(string transactionPath, string message)
    : MiiException(message, transactionPath, "TerminateTransaction")
{
    public string TransactionPath { get; } = transactionPath;
}

/// <summary>A value that keeps the MII document it was read from (all rowsets, all columns, MII's text).</summary>
public interface IMiiDocument
{
    /// <summary>The document, or null when the value has changed since it was read.</summary>
    XmlDocument MiiDocument { get; }
}

/// <summary>Row records read from an MII document: the rows of every rowset, and the document itself, so that
/// XPaths such as /Rowsets/Rowset[2]/Row and columns the record does not know still see what MII returned.</summary>
public sealed class MiiRowList<T> : List<T>, IMiiDocument where T : MiiRow
{
    private readonly XmlDocument document;
    private readonly int count;

    public MiiRowList(IEnumerable<T> rows, XmlDocument document) : base(rows)
    {
        this.document = document;
        count = Count;
    }

    public XmlDocument MiiDocument => Count == count ? document : null;
}

/// <summary>Base of the row records: one property per column of an MII rowset document.</summary>
public abstract record MiiRow
{
    public sealed override string ToString() => Mii.ToStr(this);
}

/// <summary>Row access as MII's XPath does it: a missing row reads as an empty row ("" for every column).</summary>
public static class MiiRows
{
    /// <summary>Rowsets/Rowset/Row/Column: the first row, or an empty row.</summary>
    public static T FirstOrEmpty<T>(this List<T> rows) where T : MiiRow, new() => rows.Count > 0 ? rows[0] : new T();

    /// <summary>Rowsets/Rowset/Row[Column = 'value']/...: the first matching row, or an empty row.</summary>
    public static T FirstOrEmpty<T>(this List<T> rows, Func<T, bool> match) where T : MiiRow, new() => rows.FirstOrDefault(match) ?? new T();

    /// <summary>Rowsets/Rowset/Row[n]/...: the row at a 1-based position, or an empty row.</summary>
    public static T RowAt<T>(this List<T> rows, int position) where T : MiiRow, new() =>
        position >= 1 && position <= rows.Count ? rows[position - 1] : new T();
}

/// <summary>Stand-ins for the MII expression language and XML handling. MII rules kept: a boolean as text is 1 or 0;
/// text is true for true/yes/on/1/y/t and false for false/no/off/0/n/f; match() ignores case unless the pattern
/// says (?-i); == converts the right side to the type of the left side; &lt; and &gt; compare numbers only; a number
/// as text has no exponent and at most 12 decimals, not-a-number is NA; datediff() returns seconds; append() changes
/// the list it gets; a map lists its keys in hash order.</summary>
public static class Mii
{
    private static readonly CultureInfo Invariant = CultureInfo.InvariantCulture;
    private static readonly Regex TrueValue = new Regex(@"^\s*(true|1)\s*$", RegexOptions.IgnoreCase);
    private static readonly Regex EmptyValue = new Regex(@"^\s*(null|---)?\s*$", RegexOptions.IgnoreCase);
    private static readonly string[] TrueWords = { "true", "yes", "on", "1", "y", "t" };
    private static readonly string[] FalseWords = { "false", "no", "off", "0", "n", "f" };

    // MII reads a date text only in the XML date format (text after the seconds is ignored); the other formats are
    // kept for values that reach the C# from outside MII
    private static readonly string[] DateFormats =
    {
        "yyyy-MM-dd'T'HH:mm:ss", "yyyy-MM-dd HH:mm:ss", "yyyy-MM-dd", "MM/dd/yyyy HH:mm:ss"
    };

    private static readonly string RunId = Guid.NewGuid().ToString();

    public static string UserName
    {
        get { return Environment.UserName; }
    }

    public static string TransactionId
    {
        get { return RunId; }
    }

    public static string UserLocale
    {
        get { return CultureInfo.CurrentCulture.Name; }
    }

    // ------------------------------------------------------------------ conversions

    public static string ToStr(object value)
    {
        if (value == null)
        {
            return string.Empty;
        }

        string text = value as string;

        if (text != null)
        {
            return text;
        }

        if (value is bool)
        {
            // a boolean as text is 1 or 0 in MII (booleantostring gives true / false)
            return (bool)value ? "1" : "0";
        }

        if (value is DateTime)
        {
            // no time (a NULL date of a query, an unset dateTime): MII's text TimeUnavailable
            return (DateTime)value == DateTime.MinValue ? "TimeUnavailable" : ((DateTime)value).ToString("yyyy-MM-dd'T'HH:mm:ss", Invariant);
        }

        if (value is double)
        {
            return NumberText((double)value);
        }

        if (value is float)
        {
            return NumberText((double)(float)value);
        }

        if (value is XmlDocument)
        {
            XmlDocument document = (XmlDocument)value;
            return document.DocumentElement == null ? string.Empty : document.DocumentElement.OuterXml;
        }

        if (value is XmlNode)
        {
            return ((XmlNode)value).OuterXml;
        }

        if (value is IMiiDocument carrier && carrier.MiiDocument != null && carrier.MiiDocument.DocumentElement != null)
        {
            return carrier.MiiDocument.DocumentElement.OuterXml;
        }

        if (value is IEnumerable<MiiRow> rows)
        {
            return RowsetDocument(rows, value.GetType()).DocumentElement.OuterXml;
        }

        if (value is MiiRow)
        {
            return RowElement(new XmlDocument(), value).OuterXml;
        }

        if (value is Exception)
        {
            return ((Exception)value).Message;
        }

        IDictionary<string, string> map = value as IDictionary<string, string>;

        if (map != null)
        {
            // MII's text of a map: {Site=[string:1000], Sfc=[string:SFC-1]} in hash order, "" when it is empty
            List<string> pairs = new List<string>();

            foreach (string key in HashOrder(map.Keys))
            {
                pairs.Add(key + "=[string:" + map[key] + "]");
            }

            return pairs.Count == 0 ? string.Empty : "{" + string.Join(", ", pairs.ToArray()) + "}";
        }

        List<string> list = value as List<string>;

        if (list != null)
        {
            // MII's text of a list: [[string:a], [string:b]], "" when it is empty
            return list.Count == 0 ? string.Empty : "[" + string.Join(", ", list.Select((item) => "[string:" + item + "]").ToArray()) + "]";
        }

        IFormattable formattable = value as IFormattable;

        if (formattable != null)
        {
            return formattable.ToString(null, Invariant);
        }

        return value.ToString();
    }

    /// <summary>A floating-point number as MII writes it: no grouping, no exponent, at most 12 decimals (rounded
    /// half-even), 12.0 as 12. Not-a-number is NA; so is the result of a division by zero (infinity in C#).</summary>
    public static string NumberText(double value)
    {
        if (double.IsNaN(value) || double.IsInfinity(value))
        {
            return "NA";
        }

        string digits;
        decimal exact;

        if (Math.Abs(value) < 7.9e27 && decimal.TryParse(value.ToString("R", Invariant), NumberStyles.Float, Invariant, out exact))
        {
            digits = Math.Abs(Math.Round(exact, 12, MidpointRounding.ToEven)).ToString("0.############", Invariant);
        }
        else
        {
            digits = Math.Abs(value) < 1 ? "0" : Math.Abs(value).ToString("F0", Invariant);
        }

        // Java writes the sign also when nothing is left of the value (-0)
        return value < 0 ? "-" + digits : digits;
    }

    /// <summary>The keys of a map in the order MII lists them: the order of a Java HashMap (by hash bucket, neither the
    /// order of insertion nor sorted) for a map that was filled entry by entry.</summary>
    public static List<string> HashOrder(IEnumerable<string> keys)
    {
        List<string> all = new List<string>(keys);
        int capacity = 16;

        while (all.Count > capacity / 4 * 3)
        {
            capacity *= 2;
        }

        List<KeyValuePair<int, string>> buckets = new List<KeyValuePair<int, string>>();

        foreach (string key in all)
        {
            int hash = 0;

            foreach (char c in key)
            {
                hash = unchecked(31 * hash + c);
            }

            hash ^= (int)((uint)hash >> 16);
            buckets.Add(new KeyValuePair<int, string>(hash & (capacity - 1), key));
        }

        // a stable sort: keys of one bucket stay in the order they were put
        return buckets.OrderBy((entry) => entry.Key).Select((entry) => entry.Value).ToList();
    }

    public static bool ToBool(object value)
    {
        if (value is bool)
        {
            return (bool)value;
        }

        if (value is int || value is long || value is double || value is decimal)
        {
            return Convert.ToDouble(value, Invariant) != 0;
        }

        bool result = false;
        string text = ToStr(value);

        // "" stands for a boolean that is not set (an output a service did not fill); in MII an empty text fails too
        if (text.Length == 0 || TryBool(text, out result))
        {
            return text.Length > 0 && result;
        }

        throw new FormatException("Conversion Exception: \"" + text + "\" is no boolean (MII reads true/yes/on/1/y/t, false/no/off/0/n/f and whole numbers, without blanks around)");
    }

    /// <summary>Text as a boolean the way MII converts it: the words true/yes/on/1/y/t and false/no/off/0/n/f in any
    /// case, a whole number (1 or more is true), a text with a decimal point is false; anything else is no boolean.</summary>
    private static bool TryBool(string text, out bool result)
    {
        string lower = text.ToLowerInvariant();
        result = Array.IndexOf(TrueWords, lower) >= 0;

        if (result || Array.IndexOf(FalseWords, lower) >= 0)
        {
            return true;
        }

        long whole;
        double number;

        if (long.TryParse(text, NumberStyles.AllowLeadingSign, Invariant, out whole))
        {
            result = whole >= 1;
            return true;
        }

        return text.Contains(".") && double.TryParse(text, NumberStyles.Float, Invariant, out number);
    }

    /// <summary>stringtoboolean(text): true only for the exact text true.</summary>
    public static bool StringToBoolean(object value)
    {
        return value is bool ? (bool)value : ToStr(value) == "true";
    }

    /// <summary>The operand of a comparison (&lt; &gt; &lt;= &gt;=) as MII reads it: always a number. "" and NA are
    /// not-a-number (every comparison with it is false), a date counts by its instant, other text fails.</summary>
    public static double ToNumber(object value)
    {
        if (value is DateTime)
        {
            return ((DateTime)value - new DateTime(1970, 1, 1)).TotalMilliseconds;
        }

        if (value == null || IsNumber(value) || value is bool)
        {
            return value == null ? double.NaN : ToDouble(value);
        }

        string text = ToStr(value).Trim();
        double result;

        if (text.Length == 0 || text == "NA")
        {
            return double.NaN;
        }

        if (double.TryParse(text, NumberStyles.Float, Invariant, out result))
        {
            return result;
        }

        throw new FormatException("Could not evaluate operator node: \"" + text + "\" is no number (MII compares numbers only)");
    }

    public static double ToDouble(object value)
    {
        if (value is double)
        {
            return (double)value;
        }

        if (value is int || value is long || value is decimal || value is float)
        {
            return Convert.ToDouble(value, Invariant);
        }

        if (value is bool)
        {
            return (bool)value ? 1 : 0;
        }

        double result;
        return double.TryParse(ToStr(value).Trim(), NumberStyles.Float, Invariant, out result) ? result : 0;
    }

    private static readonly Regex LeadingNumber = new Regex(@"^-?(?:\d[\d,]*(?:\.\d*)?|\.\d+)(?:E-?\d+)?");

    /// <summary>number(x) reads the number a text starts with and ignores the rest: number("12abc") is 12,
    /// number("1,5") is 15 (the comma groups), number("1e3") is 1 (only an upper-case E is an exponent), number(" 5 ")
    /// is 5. "---" is not-a-number; "" and a text without a leading number fail (measured on MII).</summary>
    public static double Number(object value)
    {
        if (value is double || value is int || value is long || value is decimal || value is float || value is bool)
        {
            return ToDouble(value);
        }

        string text = ToStr(value);
        double result;

        if (value == null || text == "---")
        {
            return double.NaN;
        }

        Match leading = LeadingNumber.Match(text);

        if (leading.Success && double.TryParse(leading.Value.Replace(",", string.Empty), NumberStyles.Float, Invariant, out result))
        {
            return result;
        }

        // no leading number: the whole text, blanks around it removed
        if (text.Trim().Length > 0 && double.TryParse(text.Trim(), NumberStyles.Float, Invariant, out result))
        {
            return result;
        }

        throw new FormatException("Unable to convert the the value to a number. (number(\"" + text + "\"))");
    }

    /// <summary>if(condition, a, b) returns a number in MII, also for text branches: text that is no number gives 0
    /// (assigned to a string) or a type cast error (inside a string function); stringif returns text.</summary>
    public static double If(bool condition, object whenTrue, object whenFalse)
    {
        return If(condition ? whenTrue : whenFalse);
    }

    /// <summary>The result of if(): the chosen branch as a number, 0 for text that is no number.</summary>
    public static double If(object chosen)
    {
        double result;
        return TryNumber(chosen, out result) ? result : 0;
    }

    public static int ToInt(object value)
    {
        double number = ToDouble(value);
        return number > int.MaxValue ? int.MaxValue : number < int.MinValue ? int.MinValue : (int)number;
    }

    public static long ToLong(object value)
    {
        if (value is long)
        {
            return (long)value;
        }

        long result;

        if (long.TryParse(ToStr(value).Trim(), NumberStyles.Integer, Invariant, out result))
        {
            return result;
        }

        return (long)ToDouble(value);
    }

    public static decimal ToDecimal(object value)
    {
        if (value is decimal)
        {
            return (decimal)value;
        }

        decimal result;

        if (decimal.TryParse(ToStr(value).Trim(), NumberStyles.Float, Invariant, out result))
        {
            return result;
        }

        double number = ToDouble(value);
        return number > (double)decimal.MaxValue || number < (double)decimal.MinValue ? 0m : (decimal)number;
    }

    public static DateTime ToDateTime(object value)
    {
        if (value is DateTime)
        {
            return (DateTime)value;
        }

        DateTime result;
        string text = ToStr(value).Trim();

        // MII ignores what follows the seconds (milliseconds, Z, an offset): the text is taken as local time
        if (text.Length > 19 && text[10] == 'T' && DateTime.TryParseExact(text.Substring(0, 19), DateFormats[0], Invariant, DateTimeStyles.None, out result))
        {
            return result;
        }

        if (DateTime.TryParseExact(text, DateFormats, Invariant, DateTimeStyles.None, out result))
        {
            return result;
        }

        return DateTime.MinValue;
    }

    public static XmlDocument ToXml(object value)
    {
        XmlDocument document = value as XmlDocument;

        if (document != null)
        {
            return document;
        }

        if (value is IMiiDocument carrier && carrier.MiiDocument != null)
        {
            return (XmlDocument)carrier.MiiDocument.CloneNode(true);
        }

        if (value is IEnumerable<MiiRow> rows)
        {
            return RowsetDocument(rows, value.GetType());
        }

        if (value is MiiRow)
        {
            XmlDocument rowDocument = new XmlDocument();
            rowDocument.AppendChild(RowElement(rowDocument, value));
            return rowDocument;
        }

        XmlNode node = value as XmlNode;

        if (node != null)
        {
            return NodeAsDocument(node);
        }

        string text = ToStr(value).Trim();
        return text.StartsWith("<") ? ParseXml(text) : new XmlDocument();
    }

    public static List<string> ToList(object value)
    {
        List<string> list = value as List<string>;

        if (list != null)
        {
            return list;
        }

        List<string> result = new List<string>();

        if (value == null)
        {
            return result;
        }

        IDictionary<string, string> map = value as IDictionary<string, string>;

        if (map != null)
        {
            // an Iterator over a map goes through its keys, in the map's hash order
            result.AddRange(HashOrder(map.Keys));
            return result;
        }

        System.Collections.IEnumerable items = value as System.Collections.IEnumerable;

        if (items != null && !(value is string) && !(value is XmlNode))
        {
            foreach (object item in items)
            {
                result.Add(ToStr(item));
            }

            return result;
        }

        string text = ToStr(value).Trim();

        if (text.StartsWith("[") && text.EndsWith("]"))
        {
            foreach (Match match in Regex.Matches(text, @"\[(?:[A-Za-z]+:)?([^\[\]]*)\]"))
            {
                result.Add(match.Groups[1].Value);
            }

            return result;
        }

        if (text.Length > 0)
        {
            result.Add(text);
        }

        return result;
    }

    public static Dictionary<string, string> ToMap(object value)
    {
        Dictionary<string, string> map = value as Dictionary<string, string>;

        if (map != null)
        {
            return map;
        }

        Dictionary<string, string> result = new Dictionary<string, string>();
        IDictionary<string, string> other = value as IDictionary<string, string>;

        if (other != null)
        {
            foreach (KeyValuePair<string, string> pair in other)
            {
                result[pair.Key] = pair.Value;
            }

            return result;
        }

        foreach (Match match in Regex.Matches(ToStr(value), @"([^{},=\s][^{},=]*)=\[(?:[A-Za-z]+:)?([^\[\]]*)\]"))
        {
            result[match.Groups[1].Value.Trim()] = match.Groups[2].Value;
        }

        return result;
    }

    /// <summary>An MII rowset document (or anything that converts to one) as row records, columns matched by name.</summary>
    public static List<T> ToRows<T>(object value) where T : MiiRow, new()
    {
        if (value is List<T> list)
        {
            return list;
        }

        List<T> rows = [];
        XmlDocument document = ToXml(value);

        if (document.DocumentElement == null)
        {
            return rows;
        }

        foreach (XmlNode node in document.SelectNodes("/Rowsets/Rowset/Row"))
        {
            T row = new T();
            FillRow(row, node);
            KeepRowSource(row, (XmlElement)node);
            rows.Add(row);
        }

        // the rows keep the document: columns the record does not know and further rowsets stay readable
        return new MiiRowList<T>(rows, value is XmlDocument ? (XmlDocument)document.CloneNode(true) : document);
    }

    /// <summary>Row records as an MII rowset document (Rowsets/Rowset with Columns and Row elements).</summary>
    private static XmlDocument RowsetDocument(IEnumerable<MiiRow> rows, Type listType)
    {
        XmlDocument document = new XmlDocument();
        XmlElement rowset = document.CreateElement("Rowset");
        document.AppendChild(document.CreateElement("Rowsets")).AppendChild(rowset);
        XmlElement columns = document.CreateElement("Columns");
        rowset.AppendChild(columns);
        Type rowType = listType.IsGenericType ? listType.GetGenericArguments()[0] : typeof(MiiRow);

        foreach (PropertyInfo property in ColumnsOf(rowType))
        {
            XmlElement column = document.CreateElement("Column");
            column.SetAttribute("Name", property.Name);
            columns.AppendChild(column);
        }

        foreach (MiiRow row in rows)
        {
            rowset.AppendChild(RowElement(document, row));
        }

        return document;
    }

    public static T ToRow<T>(object value) where T : MiiRow, new()
    {
        if (value is T)
        {
            return (T)value;
        }

        T row = new T();
        XmlDocument document = ToXml(value);
        XmlNode node = document.SelectSingleNode("//Row") ?? document.DocumentElement;

        if (node != null)
        {
            FillRow(row, node);
        }

        return row;
    }

    // ------------------------------------------------------------------ comparisons

    /// <summary>== in MII: the right side is converted to the type of the left side, then both are compared. So
    /// 5 == "5.0" is true and "5.0" == 5 is false ("5.0" against "5"); "true" == true is false (a boolean as text is
    /// 1). A right side that cannot be converted makes the two unequal; == never fails.</summary>
    public static bool AreEqual(object left, object right)
    {
        double number;

        if (left is int || left is long)
        {
            // a whole number on the left: the right side is cut to a whole number ("2.7" equals 2)
            return TryNumber(right, out number) && !double.IsNaN(number) && Convert.ToInt64(left, Invariant) == (long)number;
        }

        if (IsNumber(left))
        {
            return TryNumber(right, out number) && Convert.ToDouble(left, Invariant) == number;
        }

        if (left is bool)
        {
            bool flag;

            if (right is bool || IsNumber(right))
            {
                return (bool)left == ToBool(right);
            }

            return TryBool(ToStr(right), out flag) && (bool)left == flag;
        }

        if (left is DateTime)
        {
            DateTime date = ToDateTime(right);
            return (right is DateTime || date != DateTime.MinValue || ToStr(right) == "TimeUnavailable") && ToStr(left) == ToStr(date);
        }

        return ToStr(left) == ToStr(right);
    }

    /// <summary>&lt; &gt; in MII compare numbers only (files of X-Ray before 2.8 call this; newer ones compare
    /// Mii.ToNumber(a) with Mii.ToNumber(b), which also gives false for an empty side).</summary>
    public static int Compare(object left, object right)
    {
        double a = ToNumber(left);
        double b = ToNumber(right);
        return double.IsNaN(a) || double.IsNaN(b) ? 0 : a.CompareTo(b);
    }

    /// <summary>&amp;&amp; with a text on the left: MII does not convert to booleans, it is true only when the right
    /// side is a text with exactly the same content ("false" &amp;&amp; "false" is true, a text &amp;&amp; true is
    /// false).</summary>
    public static bool AndText(object left, object right)
    {
        if (left is string)
        {
            return right is string && (string)left == (string)right;
        }

        return left != null && right != null && ToBool(left) & ToBool(right);
    }

    private static bool IsNumber(object value)
    {
        return value is int || value is long || value is double || value is decimal || value is float;
    }

    private static bool TryNumber(object value, out double number)
    {
        if (IsNumber(value))
        {
            number = Convert.ToDouble(value, Invariant);
            return true;
        }

        return double.TryParse(ToStr(value).Trim(), NumberStyles.Float, Invariant, out number);
    }

    // ------------------------------------------------------------------ XML

    public static XmlDocument ParseXml(string text)
    {
        XmlDocument document = new XmlDocument();

        if (text != null && text.Trim().Length > 0)
        {
            document.LoadXml(text.Trim());
        }

        return document;
    }

    /// <summary>Doc{xpath}: the text of the first matching node, or the value of an XPath expression such as count(...).</summary>
    public static string XPath(object source, string xpath)
    {
        XmlDocument document = ToXml(source);

        if (document.DocumentElement == null)
        {
            return string.Empty;
        }

        object result = EvaluateXPath(document, xpath);
        System.Xml.XPath.XPathNodeIterator nodes = result as System.Xml.XPath.XPathNodeIterator;

        if (nodes != null)
        {
            return nodes.MoveNext() ? nodes.Current.Value : string.Empty;
        }

        return XPathText(result);
    }

    /// <summary>Doc{xpath} used as XML: the first matching node as a document of its own.</summary>
    public static XmlDocument SelectXml(object source, string xpath)
    {
        XmlDocument document = ToXml(source);

        if (document.DocumentElement == null)
        {
            return new XmlDocument();
        }

        List<XmlNode> nodes = XPathNodes(EvaluateXPath(document, xpath));
        return nodes.Count == 0 ? new XmlDocument() : NodeAsDocument(nodes[0]);
    }

    private static readonly Regex TrailingSlash = new Regex(@"(?<=[\w\]\*\)'""])/+(?=\s*(\)|\||,|$))");

    /// <summary>An XPath 1.0 expression as MII evaluates it: a node-set, a number, a text or a boolean. A location path
    /// that ends with "/" (count(/Rowsets/Rowset/Row/)) is read as the path without it (assumed: MII runs such links).</summary>
    private static object EvaluateXPath(XmlDocument document, string xpath)
    {
        return document.CreateNavigator().Evaluate(TrailingSlash.Replace(xpath ?? string.Empty, string.Empty), Namespaces(document));
    }

    /// <summary>The nodes of an XPath result (none for a number, text or boolean).</summary>
    private static List<XmlNode> XPathNodes(object result)
    {
        List<XmlNode> nodes = new List<XmlNode>();
        System.Xml.XPath.XPathNodeIterator iterator = result as System.Xml.XPath.XPathNodeIterator;

        while (iterator != null && iterator.MoveNext())
        {
            System.Xml.IHasXmlNode node = iterator.Current as System.Xml.IHasXmlNode;

            if (node != null)
            {
                nodes.Add(node.GetNode());
            }
        }

        return nodes;
    }

    /// <summary>A number, text or boolean result of XPath as text (Java's XPath string form: 2, 0.5, NaN, Infinity).</summary>
    private static string XPathText(object result)
    {
        if (result is double)
        {
            double number = (double)result;

            if (double.IsNaN(number))
            {
                return "NaN";
            }

            if (double.IsInfinity(number))
            {
                return number > 0 ? "Infinity" : "-Infinity";
            }

            return number == Math.Floor(number) && Math.Abs(number) < 1e15 ? ((long)number).ToString(Invariant) : number.ToString("R", Invariant);
        }

        return ToStr(result);
    }

    /// <summary>The nodes a repeater runs over, each as a document of its own.</summary>
    public static List<XmlDocument> SelectNodes(object source, string xpath)
    {
        List<XmlDocument> result = new List<XmlDocument>();
        XmlDocument document = ToXml(source);

        if (document.DocumentElement == null)
        {
            return result;
        }

        foreach (XmlNode node in XPathNodes(EvaluateXPath(document, xpath)))
        {
            result.Add(NodeAsDocument(node));
        }

        return result;
    }

    /// <summary>X{key} on a value of unknown type: a map lookup, else an XPath.</summary>
    public static string Select(object source, string key)
    {
        IDictionary<string, string> map = source as IDictionary<string, string>;

        if (map != null)
        {
            return MapGet(map, key);
        }

        return XPath(source, key);
    }

    /// <summary>Assign link with XPath: sets the text of the selected node. The node must exist, MII never creates it
    /// ("No nodes found ..."); only in a document that is still empty X-Ray builds a plain element path.</summary>
    public static XmlDocument SetXPathValue(XmlDocument document, string xpath, object value)
    {
        XmlNode node = FindOrCreate(document, xpath);

        if (node != null)
        {
            node.InnerText = ToStr(value);
        }

        return document;
    }

    /// <summary>Assign XML link: without XPath the source document replaces the whole document; at an XPath the selected
    /// element is replaced by a copy of the source element, so the source's element name wins (measured on MII:
    /// Request{/Envelope/Body} from Template{/Template} gives Envelope/Template). A source that is no XML has no effect.</summary>
    public static XmlDocument AssignXml(XmlDocument document, string xpath, object source)
    {
        XmlDocument sourceDocument = ToXml(source);

        if (sourceDocument.DocumentElement == null)
        {
            return document;
        }

        XmlNode node = FindOrCreate(document, xpath);

        if (node == null || node == document.DocumentElement || node == document || node.ParentNode == null)
        {
            document.RemoveAll();
            document.AppendChild(document.ImportNode(sourceDocument.DocumentElement, true));
            return document;
        }

        node.ParentNode.ReplaceChild(document.ImportNode(sourceDocument.DocumentElement, true), node);
        return document;
    }

    /// <summary>Append XML link: the source root element becomes the last child of the selected node.</summary>
    public static XmlDocument AppendXml(XmlDocument document, string xpath, object source)
    {
        XmlNode node = FindOrCreate(document, xpath);

        foreach (XmlElement element in SourceElements(source))
        {
            if (node != null)
            {
                node.AppendChild(document.ImportNode(element, true));
            }
        }

        return document;
    }

    /// <summary>Append After XML link: the source root element is inserted after the selected node.</summary>
    public static XmlDocument AppendAfterXml(XmlDocument document, string xpath, object source)
    {
        XmlNode node = FindOrCreate(document, xpath);

        foreach (XmlElement element in SourceElements(source))
        {
            if (node != null && node.ParentNode != null)
            {
                node = node.ParentNode.InsertAfter(document.ImportNode(element, true), node);
            }
        }

        return document;
    }

    /// <summary>The elements an XML link appends: the root of the source, or each node an XPath selected
    /// (Doc{xpath} as the source of Append XML).</summary>
    private static List<XmlElement> SourceElements(object source)
    {
        List<XmlElement> elements = new List<XmlElement>();

        if (source is IEnumerable<XmlDocument> nodes)
        {
            foreach (XmlDocument item in nodes)
            {
                if (item.DocumentElement != null)
                {
                    elements.Add(item.DocumentElement);
                }
            }

            return elements;
        }

        XmlDocument sourceDocument = ToXml(source);

        if (sourceDocument.DocumentElement != null)
        {
            elements.Add(sourceDocument.DocumentElement);
        }

        return elements;
    }

    /// <summary>Replace XML link: the children of the selected node are replaced by the source root element.</summary>
    public static XmlDocument ReplaceXml(XmlDocument document, string xpath, object source)
    {
        XmlDocument sourceDocument = ToXml(source);
        XmlNode node = FindOrCreate(document, xpath);

        if (node != null && sourceDocument.DocumentElement != null)
        {
            while (node.FirstChild != null)
            {
                node.RemoveChild(node.FirstChild);
            }

            node.AppendChild(document.ImportNode(sourceDocument.DocumentElement, true));
        }

        return document;
    }

    /// <summary>Remove XML link: removes every node the XPath selects.</summary>
    public static XmlDocument RemoveXml(XmlDocument document, string xpath)
    {
        if (document.DocumentElement == null)
        {
            return document;
        }

        List<XmlNode> nodes = new List<XmlNode>();

        foreach (XmlNode node in document.SelectNodes(xpath, Namespaces(document)))
        {
            nodes.Add(node);
        }

        foreach (XmlNode node in nodes)
        {
            if (node.ParentNode != null)
            {
                node.ParentNode.RemoveChild(node);
            }
        }

        return document;
    }

    /// <summary>XSLT action.</summary>
    public static XmlDocument XslTransform(object input, object stylesheet)
    {
        XslCompiledTransform transform = new XslCompiledTransform();

        using (XmlReader reader = new XmlNodeReader(ToXml(stylesheet)))
        {
            transform.Load(reader);
        }

        StringBuilder output = new StringBuilder();

        using (StringWriter writer = new StringWriter(output, Invariant))
        {
            transform.Transform(ToXml(input), null, writer);
        }

        string text = output.ToString().Trim();
        return text.StartsWith("<") ? ParseXml(text) : new XmlDocument();
    }

    /// <summary>A row for a rowset document that is not typed: column name, value, column name, value, ...</summary>
    public static void AddRow(object document, params object[] columnsAndValues)
    {
        XmlDocument xml = ToXml(document);
        XmlNode rowset = xml.SelectSingleNode("/Rowsets/Rowset");

        if (rowset == null)
        {
            return;
        }

        XmlElement row = xml.CreateElement("Row");

        for (int index = 0; index + 1 < columnsAndValues.Length; index += 2)
        {
            XmlElement cell = xml.CreateElement(ToStr(columnsAndValues[index]));
            cell.InnerText = ToStr(columnsAndValues[index + 1]);
            row.AppendChild(cell);
        }

        rowset.AppendChild(row);
    }

    public static List<string> GetListFromXPath(object source, object xpath)
    {
        List<string> result = new List<string>();
        XmlDocument document = ToXml(source);

        if (document.DocumentElement == null)
        {
            return result;
        }

        object value = EvaluateXPath(document, ToStr(xpath));

        if (!(value is System.Xml.XPath.XPathNodeIterator))
        {
            // getlistfromxpath(doc, "string(...)") or count(...): one element with the value
            result.Add(XPathText(value));
            return result;
        }

        foreach (XmlNode node in XPathNodes(value))
        {
            result.Add(node.InnerText);
        }

        return result;
    }

    public static Dictionary<string, string> GetMapFromXPath(object source, object xpath)
    {
        Dictionary<string, string> result = new Dictionary<string, string>();
        XmlDocument document = ToXml(source);

        if (document.DocumentElement == null)
        {
            return result;
        }

        foreach (XmlNode node in XPathNodes(EvaluateXPath(document, ToStr(xpath))))
        {
            result[node.Name] = node.InnerText;
        }

        return result;
    }

    public static string Name(object source)
    {
        XmlDocument document = ToXml(source);
        return document.DocumentElement == null ? string.Empty : document.DocumentElement.Name;
    }

    private static XmlNamespaceManager Namespaces(XmlDocument document)
    {
        XmlNamespaceManager manager = new XmlNamespaceManager(document.NameTable);

        foreach (XmlNode node in document.SelectNodes("//namespace::*"))
        {
            string prefix = node.LocalName == "xmlns" ? string.Empty : node.LocalName;

            if (prefix.Length > 0 && prefix != "xml" && manager.LookupNamespace(prefix) == null)
            {
                manager.AddNamespace(prefix, node.Value);
            }
        }

        return manager;
    }

    private static XmlDocument NodeAsDocument(XmlNode node)
    {
        XmlDocument document = new XmlDocument();

        if (node is XmlDocument)
        {
            XmlNode root = ((XmlDocument)node).DocumentElement;

            if (root != null)
            {
                document.AppendChild(document.ImportNode(root, true));
            }

            return document;
        }

        if (node is XmlElement)
        {
            XmlElement copy = (XmlElement)document.AppendChild(document.ImportNode(node, true));

            // the namespace declarations in scope of the node go with it (a repeater item keeps its prefixes)
            foreach (KeyValuePair<string, string> scope in node.CreateNavigator().GetNamespacesInScope(System.Xml.XmlNamespaceScope.ExcludeXml))
            {
                string name = scope.Key.Length == 0 ? "xmlns" : "xmlns:" + scope.Key;

                if (scope.Value.Length > 0 && !copy.HasAttribute(name))
                {
                    XmlAttribute declaration = document.CreateAttribute(name, "http://www.w3.org/2000/xmlns/");
                    declaration.Value = scope.Value;
                    copy.Attributes.Append(declaration);
                }
            }

            return document;
        }

        XmlElement value = document.CreateElement("Value");
        value.InnerText = node.Value ?? node.InnerText;
        document.AppendChild(value);
        return document;
    }

    private static readonly System.Runtime.CompilerServices.ConditionalWeakTable<XmlDocument, object> BuiltDocuments = new System.Runtime.CompilerServices.ConditionalWeakTable<XmlDocument, object>();

    /// <summary>The node an XML link writes to. In MII the node must exist: an XPath that selects nothing fails the
    /// link ("No nodes found in xpath ..."). A document that X-Ray starts empty is the exception: there a plain element
    /// path (/a/b/c) is built, also by the links that follow (MII starts such a document with content, for example the
    /// input of a called transaction with the callee's default).</summary>
    private static XmlNode FindOrCreate(XmlDocument document, string xpath)
    {
        object built;
        bool plainPath = Regex.IsMatch(xpath, @"^(/[A-Za-z_][\w.\-]*)+$");

        if (document.DocumentElement != null)
        {
            XmlNode found = document.SelectSingleNode(xpath, Namespaces(document));

            if (found != null)
            {
                return found;
            }

            if (!plainPath || !BuiltDocuments.TryGetValue(document, out built))
            {
                throw new InvalidOperationException("No nodes found in xpath: \"" + xpath + "\" XML was:" + document.DocumentElement.OuterXml);
            }
        }

        if (!plainPath)
        {
            return null;
        }

        if (!BuiltDocuments.TryGetValue(document, out built))
        {
            BuiltDocuments.Add(document, string.Empty);
        }

        XmlNode current = document;

        foreach (string name in xpath.Trim('/').Split('/'))
        {
            XmlNode next = null;

            foreach (XmlNode child in current.ChildNodes)
            {
                if (child.NodeType == XmlNodeType.Element && child.Name == name)
                {
                    next = child;
                    break;
                }
            }

            if (next == null)
            {
                if (current == document && document.DocumentElement != null)
                {
                    return null;
                }

                next = document.CreateElement(name);
                current.AppendChild(next);
            }

            current = next;
        }

        return current;
    }

    // ------------------------------------------------------------------ rows (reflection over the generated row classes)

    public static List<PropertyInfo> ColumnsOf(Type rowType)
    {
        List<PropertyInfo> result = new List<PropertyInfo>();

        foreach (PropertyInfo property in rowType.GetProperties(BindingFlags.Public | BindingFlags.Instance))
        {
            if (property.CanRead && property.CanWrite && property.GetIndexParameters().Length == 0)
            {
                result.Add(property);
            }
        }

        return result;
    }

    public static object RowValue(object row, string column)
    {
        PropertyInfo property = FindColumn(row.GetType(), column);
        return property == null ? null : property.GetValue(row, null);
    }

    private static readonly System.Runtime.CompilerServices.ConditionalWeakTable<object, XmlElement> RowSources = new System.Runtime.CompilerServices.ConditionalWeakTable<object, XmlElement>();

    /// <summary>The Row element a record was read from: the record stands for it (columns the record does not know,
    /// e.g. of select *, included).</summary>
    public static void KeepRowSource(MiiRow row, XmlElement element)
    {
        RowSources.AddOrUpdate(row, element);
    }

    public static XmlElement RowElement(XmlDocument document, object row)
    {
        XmlElement source;

        if (RowSources.TryGetValue(row, out source))
        {
            XmlElement copy = (XmlElement)document.ImportNode(source, true);

            // a column the code changed after reading wins over the text read
            foreach (PropertyInfo property in ColumnsOf(row.GetType()))
            {
                XmlNode cell = copy.SelectSingleNode(property.Name);
                object value = property.GetValue(row, null);

                if (cell == null && copy.ChildNodes.Cast<XmlNode>().Any((child) => Simplify(child.LocalName) == Simplify(property.Name)))
                {
                    continue;
                }

                if (cell == null ? !string.IsNullOrEmpty(ToStr(value)) : !Equals(ConvertTo(cell.InnerText, property.PropertyType), value))
                {
                    (cell ?? copy.AppendChild(document.CreateElement(property.Name))).InnerText = ToStr(value);
                }
            }

            return copy;
        }

        XmlElement element = document.CreateElement("Row");

        foreach (PropertyInfo property in ColumnsOf(row.GetType()))
        {
            XmlElement cell = document.CreateElement(property.Name);
            cell.InnerText = ToStr(property.GetValue(row, null));
            element.AppendChild(cell);
        }

        return element;
    }

    public static void FillRow(object row, XmlNode node)
    {
        foreach (XmlNode cell in node.ChildNodes)
        {
            if (cell.NodeType != XmlNodeType.Element)
            {
                continue;
            }

            PropertyInfo property = FindColumn(row.GetType(), cell.LocalName);

            if (property != null)
            {
                property.SetValue(row, ConvertTo(cell.InnerText, property.PropertyType), null);
            }
        }
    }

    private static PropertyInfo FindColumn(Type rowType, string column)
    {
        string wanted = Simplify(column);
        PropertyInfo loose = null;

        foreach (PropertyInfo property in ColumnsOf(rowType))
        {
            if (property.Name == column)
            {
                return property;
            }

            if (loose == null && Simplify(property.Name) == wanted)
            {
                loose = property;
            }
        }

        return loose;
    }

    private static string Simplify(string name)
    {
        StringBuilder result = new StringBuilder();

        foreach (char c in name)
        {
            if (char.IsLetterOrDigit(c))
            {
                result.Append(char.ToLowerInvariant(c));
            }
        }

        return result.ToString();
    }

    private static object ConvertTo(string text, Type type)
    {
        if (type == typeof(string)) return text;

        if (type == typeof(bool)) return ToBool(text);

        if (type == typeof(int)) return ToInt(text);

        if (type == typeof(long)) return ToLong(text);

        if (type == typeof(double)) return ToDouble(text);

        if (type == typeof(decimal)) return ToDecimal(text);

        if (type == typeof(DateTime)) return ToDateTime(text);
        return text;
    }

    public static List<string> List(params object[] values)
    {
        List<string> result = new List<string>();

        foreach (object value in values)
        {
            // a list argument contributes its elements: list(Local.Items, x) is a new, longer list
            List<string> items = value as List<string>;

            if (items != null)
            {
                result.AddRange(items);
            }
            else
            {
                result.Add(ToStr(value));
            }
        }

        return result;
    }

        {
            result[ToStr(keysAndValues[index])] = ToStr(keysAndValues[index + 1]);
        }

        return result;
    }

    public static int Size(object value)
    {
        IDictionary<string, string> map = value as IDictionary<string, string>;

        if (map != null)
        {
            return map.Count;
        }

        // size() of anything that is no list and no map is 0 in MII. A text in MII's list or map notation counts as
        // the collection it stands for (a service may hand a list over as text).
        string text = value as string;

        if (text != null)
        {
            text = text.Trim();
            return text.StartsWith("[") && text.EndsWith("]") ? ToList(text).Count : text.StartsWith("{") && text.EndsWith("}") ? ToMap(text).Count : 0;
        }

        return value == null || value is XmlNode ? 0 : ToList(value).Count;
    }

    /// <summary>append(list, value1, ...) works by reference in MII: the list itself gets the values (a list among

            }
        }

        return items;
    }

    public static Dictionary<string, string> ListsToMap(object keys, object values)
    {
        List<string> keyList = ToList(keys);
        List<string> valueList = ToList(values);

        if (keyList.Count != valueList.Count)
        {
            throw new InvalidOperationException("liststomap fails in MII: " + keyList.Count + " keys and " + valueList.Count + " values");
        }

        // the same kind of map as map(): its keys come in hash order, liststomap does not sort them (measured on MII)
        Dictionary<string, string> result = new Dictionary<string, string>();

        for (int index = 0; index < keyList.Count; index++)
        {
            result[keyList[index].Trim()] = valueList[index];
        }

        return result;
    }

    /// <summary>stringtolist(text, delimiter): splits on a regular expression (default ",") as Java's split does: the

    private static List<string> JavaSplit(string text, string delimiter, int limit)
    {
        List<string> parts = new List<string>();
        int start = 0;

        foreach (Match match in Regex.Matches(text, JavaPattern(delimiter)))
        {
            if (limit > 0 && parts.Count == limit - 1)
            {
                break;
            }

            // a match of no width at the very start gives no empty first part
            if (match.Length == 0 && (match.Index == 0 || match.Index >= text.Length))
            {
                continue;
            }

            parts.Add(text.Substring(start, match.Index - start));
            start = match.Index + match.Length;
        }

        parts.Add(text.Substring(start));

        while (limit == 0 && parts.Count > 1 && parts[parts.Count - 1].Length == 0)
        {
            parts.RemoveAt(parts.Count - 1);
        }

        return parts.Select((part) => part.Trim()).ToList();
    }

    /// <summary>stringreplaceall replaces every match of a regular expression. MII trims the replacement text: blanks at
    /// its start and end are lost ("$1 " inserts the group only), blanks inside stay (measured on MII). The replacement
    /// is a Java one: $1 or ${name} is a group, a backslash takes the next character as it is.</summary>
    public static string StringReplaceAll(object text, object pattern, object replacement)
    {
        string template = ToStr(replacement).Trim();
        return Regex.Replace(ToStr(text), JavaPattern(pattern), (match) => JavaReplacement(match, template));
    }

    private static string JavaReplacement(Match match, string template)
    {
        StringBuilder result = new StringBuilder();
        int index = 0;

        while (index < template.Length)
        {
            char c = template[index];

            if (c == '\\' && index + 1 < template.Length)
            {
                result.Append(template[index + 1]);
                index += 2;
            }
            else if (c == '$' && index + 1 < template.Length && template[index + 1] == '{' && template.IndexOf('}', index) > 0)
            {
                int close = template.IndexOf('}', index);
                result.Append(match.Groups[template.Substring(index + 2, close - index - 2)].Value);
                index = close + 1;
            }
            else if (c == '$' && index + 1 < template.Length && char.IsDigit(template[index + 1]))
            {
                // as many digits as still name a group of the expression
                int number = template[index + 1] - '0';
                index += 2;

                while (index < template.Length && char.IsDigit(template[index]) && number * 10 + (template[index] - '0') < match.Groups.Count)
                {
                    number = number * 10 + (template[index] - '0');
                    index++;
                }

                if (number >= match.Groups.Count)
                {
                    throw new ArgumentException("stringreplaceall: the expression has no group " + number);
                }

                result.Append(match.Groups[number].Value);
            }
            else if (c == '$')
            {
                throw new ArgumentException("stringreplaceall: a $ in the replacement must be followed by a group ($1) or be written \\$");
            }
            else
            {
                result.Append(c);
                index++;
            }
        }

        return result.ToString();
    }

    // ------------------------------------------------------------------ Java regular expressions

    private static readonly System.Collections.Concurrent.ConcurrentDictionary<string, string> JavaPatterns = new System.Collections.Concurrent.ConcurrentDictionary<string, string>();

    private static readonly Dictionary<string, string> PosixClasses = new Dictionary<string, string>
    {
        ["Lower"] = "a-z", ["Upper"] = "A-Z", ["ASCII"] = @"\x00-\x7F", ["Alpha"] = "a-zA-Z", ["Digit"] = "0-9",
        ["Alnum"] = "a-zA-Z0-9", ["Punct"] = @"!-/:-@\[-`{-~", ["Graph"] = @"!-~", ["Print"] = @" -~", ["Blank"] = @" \t",
        ["Cntrl"] = @"\x00-\x1F\x7F", ["XDigit"] = "0-9a-fA-F", ["Space"] = @" \t\n\x0B\f\r",
    };

    private const string HorizontalSpace = @" \t  ᠎ -   　";

    /// <summary>MII's regular expressions are Java patterns (java.util.regex); the Java-only forms become .NET forms:
    /// possessive quantifiers (X*+ is the atomic group (?>X*)), \x{h..h}, \Q...\E, \h, \H, \R and the POSIX classes
    /// \p{Lower} etc.</summary>
    public static string JavaPattern(object pattern)
    {
        return JavaPatterns.GetOrAdd(ToStr(pattern), ToNetPattern);
    }

    private static string ToNetPattern(string pattern)
    {
        StringBuilder result = new StringBuilder();
        Stack<int> groups = new Stack<int>();
        int atom = -1;
        int index = 0;

        while (index < pattern.Length)
        {
            char c = pattern[index];

            if (c == '\\' && index + 1 < pattern.Length)
            {
                int start = result.Length;
                index = JavaEscape(pattern, index, result, false);
                atom = start;
                continue;
            }

            if (c == '[')
            {
                int start = result.Length;
                index = JavaClass(pattern, index, result);
                atom = start;
                continue;
            }

            if (c == '(')
            {
                groups.Push(result.Length);
                result.Append(c);
                index++;
                atom = -1;
                continue;
            }

            if (c == ')')
            {
                result.Append(c);
                index++;
                atom = groups.Count > 0 ? groups.Pop() : -1;
                continue;
            }

            int quantifierEnd = c == '*' || c == '+' || c == '?' ? index + 1 : c == '{' ? QuantifierEnd(pattern, index) : index;

            if (quantifierEnd > index)
            {
                string quantifier = pattern.Substring(index, quantifierEnd - index);
                index = quantifierEnd;

                if (index < pattern.Length && pattern[index] == '+' && atom >= 0)
                {
                    string body = result.ToString(atom, result.Length - atom);
                    result.Length = atom;
                    result.Append("(?>").Append(body).Append(quantifier).Append(')');
                    index++;
                }
                else
                {
                    result.Append(quantifier);

                    if (index < pattern.Length && pattern[index] == '?')
                    {
                        result.Append('?');
                        index++;
                    }
                }

                atom = -1;
                continue;
            }

            atom = c == '|' || c == '^' || c == '$' ? -1 : result.Length;
            result.Append(c);
            index++;
        }

        return result.ToString();
    }

    private static int QuantifierEnd(string pattern, int index)
    {
        Match match = Regex.Match(pattern.Substring(index), @"^\{\d+(,\d*)?\}");
        return match.Success ? index + match.Length : index;
    }

    private static int JavaEscape(string pattern, int index, StringBuilder result, bool inClass)
    {
        char next = pattern[index + 1];

        if (next == 'x' && index + 2 < pattern.Length && pattern[index + 2] == '{')
        {
            int close = pattern.IndexOf('}', index + 3);

            if (close > 0)
            {
                int code = int.Parse(pattern.Substring(index + 3, close - index - 3), NumberStyles.HexNumber, Invariant);
                string text = char.ConvertFromUtf32(code);

                foreach (char unit in text)
                {
                    result.Append("\\u").Append(((int)unit).ToString("X4", Invariant));
                }

                return close + 1;
            }
        }

        if (next == 'Q' && !inClass)
        {
            int end = pattern.IndexOf(@"\E", index + 2, StringComparison.Ordinal);
            string text = end < 0 ? pattern.Substring(index + 2) : pattern.Substring(index + 2, end - index - 2);
            result.Append("(?:").Append(Regex.Escape(text)).Append(')');
            return end < 0 ? pattern.Length : end + 2;
        }

        if (next == 'h')
        {
            result.Append(inClass ? HorizontalSpace : "[" + HorizontalSpace + "]");
            return index + 2;
        }

        if (next == 'H' && !inClass)
        {
            result.Append("[^" + HorizontalSpace + "]");
            return index + 2;
        }

        if (next == 'R' && !inClass)
        {
            result.Append(@"(?:\r\n|[\n\x0B\f\r\u0085  ])");
            return index + 2;
        }

        if ((next == 'p' || next == 'P') && index + 2 < pattern.Length && pattern[index + 2] == '{')
        {
            int close = pattern.IndexOf('}', index + 3);
            string name = close < 0 ? string.Empty : pattern.Substring(index + 3, close - index - 3);
            string set;

            if (PosixClasses.TryGetValue(name, out set) && !(inClass && next == 'P'))
            {
                result.Append(inClass ? set : (next == 'p' ? "[" : "[^") + set + "]");
                return close + 1;
            }
        }

        if (next == '_' || (!char.IsLetterOrDigit(next) && next > 127))
        {
            // Java: a backslash before a character that is no letter quotes it (.NET refuses \_)
            result.Append(Regex.Escape(next.ToString()));
            return index + 2;
        }

        result.Append('\\').Append(next);
        return index + 2;
    }

    private static int JavaClass(string pattern, int index, StringBuilder result)
    {
        result.Append('[');
        index++;

        if (index < pattern.Length && pattern[index] == '^')
        {
            result.Append('^');
            index++;
        }

        if (index < pattern.Length && pattern[index] == ']')
        {
            result.Append(@"\]");
            index++;
        }

        while (index < pattern.Length && pattern[index] != ']')
        {
            if (pattern[index] == '\\' && index + 1 < pattern.Length)
            {
                index = JavaEscape(pattern, index, result, true);
                continue;
            }

            result.Append(pattern[index]);
            index++;
        }

        result.Append(']');
        return index + 1;
    }

    /// <summary>A number in a Java DecimalFormat pattern (0, #, '.', ',', prefix and suffix, % and per mille), rounded
    /// half-even as Java does; the value is taken as the decimal it reads as (BigDecimal). Text that is no number and
    /// an empty pattern give the value unchanged.</summary>
    public static string JavaDecimalFormat(object value, string pattern)
    {
        string text = value is double ? ((double)value).ToString("R", Invariant) : ToStr(value).Trim();
        decimal number;

        if (string.IsNullOrWhiteSpace(pattern) || !decimal.TryParse(text, NumberStyles.Float, Invariant, out number))
        {
            return text;
        }

        string positive = pattern;
        bool quoted = false;

        for (int position = 0; position < pattern.Length; position++)
        {
            if (pattern[position] == '\'')
            {
                quoted = !quoted;
            }
            else if (pattern[position] == ';' && !quoted)
            {
                positive = pattern.Substring(0, position);
                break;
            }
        }

        int first = positive.IndexOfAny("0#,.".ToCharArray());
        int last = positive.LastIndexOfAny("0#,.".ToCharArray());

        if (first < 0)
        {
            return text;
        }

        string prefix = positive.Substring(0, first);
        string body = positive.Substring(first, last - first + 1);
        string suffix = positive.Substring(last + 1);

        if ((prefix + suffix).Contains("%"))
        {
            number *= 100;
        }
        else if ((prefix + suffix).Contains("‰"))
        {
            number *= 1000;
        }

        int dot = body.IndexOf('.');
        string integerPart = dot < 0 ? body : body.Substring(0, dot);
        string fractionPart = dot < 0 ? string.Empty : body.Substring(dot + 1);
        int minimumInteger = integerPart.Count((c) => c == '0');
        int comma = integerPart.LastIndexOf(',');
        int grouping = comma < 0 ? 0 : integerPart.Length - comma - 1;
        int minimumFraction = fractionPart.Count((c) => c == '0');
        int maximumFraction = fractionPart.Count((c) => c == '0' || c == '#');
        decimal rounded = Math.Round(Math.Abs(number), Math.Min(maximumFraction, 28), MidpointRounding.ToEven);
        string digits = rounded.ToString("F" + maximumFraction, Invariant);
        string whole = maximumFraction > 0 ? digits.Substring(0, digits.IndexOf('.')) : digits;
        string fraction = maximumFraction > 0 ? digits.Substring(digits.IndexOf('.') + 1) : string.Empty;

        while (fraction.Length > minimumFraction && fraction.EndsWith("0"))
        {
            fraction = fraction.Substring(0, fraction.Length - 1);
        }

        whole = whole.TrimStart('0').PadLeft(minimumInteger, '0');

        if (whole.Length == 0 && fraction.Length == 0)
        {
            whole = "0";
        }

        if (grouping > 0)
        {
            StringBuilder grouped = new StringBuilder();

            for (int position = 0; position < whole.Length; position++)
            {
                if (position > 0 && (whole.Length - position) % grouping == 0)
                {
                    grouped.Append(',');
                }

                grouped.Append(whole[position]);
            }

            whole = grouped.ToString();
        }

        string literal = (prefix + "\u0000" + suffix).Replace("''", "\u0001").Replace("'", string.Empty).Replace("\u0001", "'");
        string[] parts = literal.Split('\u0000');
        return (number < 0 ? "-" : string.Empty) + parts[0] + whole + (fraction.Length > 0 || body.EndsWith(".") ? "." + fraction : string.Empty) + parts[1];
    }

    // MII counts seconds from the start of year 1 of its calendar: 62135769600 s before 1970-01-01 UTC
    private const long SecondsBeforeUnixEpoch = 62135769600L;

    /// <summary>datetoseconds(date): the seconds since the start of year 1 (of the instant, so the server's time zone
    /// counts); the difference of two results is the seconds between two dates.</summary>
    public static long DateToSeconds(object value)
    {
        DateTime utc = TimeZoneInfo.ConvertTimeToUtc(DateTime.SpecifyKind(ToDateTime(value), DateTimeKind.Unspecified), TimeZoneInfo.Local);
        return (long)(utc - new DateTime(1970, 1, 1)).TotalSeconds + SecondsBeforeUnixEpoch;
    }

    /// <summary>datefromseconds(seconds): the date of a datetoseconds value, in the server's time zone.</summary>
    public static DateTime DateFromSeconds(object seconds)
    {
        DateTime utc = new DateTime(1970, 1, 1, 0, 0, 0, DateTimeKind.Utc).AddSeconds(ToDouble(seconds) - SecondsBeforeUnixEpoch);
        return DateTime.SpecifyKind(TimeZoneInfo.ConvertTimeFromUtc(utc, TimeZoneInfo.Local), DateTimeKind.Unspecified);
    }

    /// <summary>converttimezone(date, toZone) converts from the server's zone, converttimezone(date, fromZone, toZone)
    /// between two zones. MII uses Java zone ids (Europe/Berlin); an id that is not known counts as GMT, as in MII.</summary>
    public static DateTime ConvertTimeZone(object value, params object[] zones)
    {
        DateTime date = ToDateTime(value);

        if (zones.Length == 0)
        {
            return date;
        }

        TimeZoneInfo source = zones.Length >= 2 ? ZoneOrGmt(ToStr(zones[0])) : TimeZoneInfo.Local;
        TimeZoneInfo target = ZoneOrGmt(ToStr(zones[zones.Length >= 2 ? 1 : 0]));
        return TimeZoneInfo.ConvertTime(DateTime.SpecifyKind(date, DateTimeKind.Unspecified), source, target);
    }

    private static TimeZoneInfo ZoneOrGmt(string id)
    {
        try
        {
            return TimeZoneInfo.FindSystemTimeZoneById(id);
        }
        catch (Exception exception) when (exception is TimeZoneNotFoundException || exception is InvalidTimeZoneException || exception is ArgumentException)
        {
            return TimeZoneInfo.Utc;
        }
    }

    /// <summary>Java date pattern -> .NET: the common letters are the same; S (milliseconds), a (AM/PM), E (day name),
    /// Z/X (offset) differ.</summary>
    private static string NetFormat(string javaFormat)
    {
        return javaFormat.Replace("SSS", "fff").Replace("a", "tt").Replace("EEEE", "dddd").Replace("EEE", "ddd").Replace("Z", "zzz").Replace("XXX", "zzz");
    }

    // ------------------------------------------------------------------ numbers

    /// <summary>round(x): the nearest whole number, a half goes up (round(-2.5) is -2). round() converts strictly: a
    /// text that is no number fails ("Conversion Exception", measured on MII), "" and NA count as 0. MII's round takes
    /// one argument; with a number of digits (not MII) it rounds to that many decimals.</summary>
    public static double Round(object value, params object[] digits)
    {
        double number;

        if (!TryNumber(value, out number))
        {
            string text = ToStr(value).Trim();

            if (text.Length > 0 && text != "NA")
            {
                throw new FormatException("Conversion Exception: round(\"" + text + "\")");
            }

            number = 0;
        }

        if (digits.Length > 0)
        {
            return Math.Round(number, Math.Max(0, Math.Min(15, ToInt(digits[0]))), MidpointRounding.AwayFromZero);
        }

        return double.IsNaN(number) ? 0 : Math.Floor(number + 0.5);
    }

    public static double Min(params object[] values)
    {
        List<double> numbers = Numbers(values);
        double result = numbers.Count > 0 ? numbers[0] : 0;

        foreach (double number in numbers)
        {
            result = Math.Min(result, number);
        }

        return result;
    }

    public static double Max(params object[] values)
    {
        List<double> numbers = Numbers(values);
        double result = numbers.Count > 0 ? numbers[0] : 0;

        foreach (double number in numbers)
        {
            result = Math.Max(result, number);
        }

        return result;
    }

    public static double Sum(params object[] values)
    {
        double result = 0;

        foreach (double number in Numbers(values))
        {
            result += number;
        }

        return result;
    }

    /// <summary>average(a, b, ...): the sum of all numbers (a list counts with its elements) divided by the number
    /// of arguments, as MII computes it.</summary>
    public static double Average(params object[] values)
    {
        return values.Length == 0 ? 0 : Sum(values) / values.Length;
    }

    private static readonly Random Randomizer = new Random();

    /// <summary>random() between 0 and 1, random(low, high) between low and high.</summary>
    public static double Random(params object[] bounds)
    {
        double low = bounds.Length > 1 ? ToDouble(bounds[0]) : 0;
        double high = bounds.Length > 1 ? ToDouble(bounds[1]) : 1;
        return low + Randomizer.NextDouble() * (high - low);
    }

    private static List<double> Numbers(object[] values)
    {
        List<double> result = new List<double>();

        foreach (object value in values)
        {
            if (value is List<string>)
            {
                foreach (string item in (List<string>)value)
                {
                    result.Add(ToDouble(item));
                }
            }
            else
            {
                result.Add(ToDouble(value));
            }
        }

        return result;
    }
